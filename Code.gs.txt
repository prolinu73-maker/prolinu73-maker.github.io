/* ---------- 화면에서 부르는 입구 ---------- */
function doGet(e) {
  var p = (e && e.parameter) || {};
  var out;
  try {
    if (p.action === 'visit') out = visits(true);          // 접속자 수 1 올리고 알려 줌
    else if (p.action === 'count') out = visits(false);    // 올리지 않고 숫자만 알려 줌
    else out = lookup(String(p.from || ''), String(p.to || ''), String(p.date || ''));
  } catch (err) {
    out = { ok: false, errors: [String(err)] };
  }
  return ContentService.createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}

/* ---------- 접속자 수 (오늘 / 누적) ---------- */
// 숫자는 이 구글 스크립트의 "스크립트 속성"에 저장됩니다. 날짜가 바뀌면 오늘 숫자는 0부터 다시 셉니다.
function visits(add) {
  var lock = LockService.getScriptLock();
  lock.waitLock(5000);   // 여러 사람이 동시에 들어와도 숫자가 꼬이지 않게 차례로 처리
  try {
    var props = PropertiesService.getScriptProperties();
    var today = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyyMMdd');
    var day = props.getProperty('visitDay');
    var n = Number(props.getProperty('visitToday')) || 0;
    var total = Number(props.getProperty('visitTotal')) || 0;
    if (day !== today) { day = today; n = 0; }
    if (add) { n++; total++; }
    props.setProperties({ visitDay: day, visitToday: String(n), visitTotal: String(total) });
    return { ok: true, today: n, total: total, date: today };
  } finally {
    lock.releaseLock();
  }
}

/* ---------- 본 처리 ---------- */
function lookup(from, to, date) {
  ERR = [];
  from = cityName(from); to = cityName(to);
  var today = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyyMMdd');
  if (!/^\d{8}$/.test(date)) date = today;
  if (!from || !to) return { ok: false, errors: ['출발지와 도착지가 필요합니다.'] };
  if (SERVICE_KEY.indexOf('여기에') === 0) return { ok: false, errors: ['중계 서버에 인증키가 입력되지 않았습니다.'] };

  var bus = [], train = [], air = [];
  try { bus = bus.concat(busOptions('exp', from, to, date)); } catch (e1) { ERR.push('고속버스: ' + e1); }
  // 시외버스는 공공데이터포털이 "당일 배차정보만" 제공하므로 오늘 날짜로 조회합니다.
  try { bus = bus.concat(busOptions('sub', from, to, today)); } catch (e2) { ERR.push('시외버스: ' + e2); }
  try { train = trainOptions(from, to, date); } catch (e3) { ERR.push('기차: ' + e3); }
  try { air = airOptions(from, to, date); } catch (e4) { ERR.push('항공기: ' + e4); }

  return { ok: true, from: from, to: to, date: date,
           bus: tidy(bus), train: tidy(train, true), air: tidy(air), errors: ERR };
}

function busOptions(kind, from, to, date) {
  var termKey = kind === 'exp' ? 'expTerm' : 'subTerm';
  var infoKey = kind === 'exp' ? 'expInfo' : 'subInfo';
  var word = kind === 'exp' ? '고속' : '시외';
  var terms = cached('terms_' + kind, function () {
    return callAll(termKey, {}).map(function (t) { return { id: t.terminalId, nm: String(t.terminalNm || '') }; });
  });
  var A = pick(terms, [from], false, 8), B = pick(terms, [to], false, 8);
  var pairs = [];
  A.forEach(function (a) { B.forEach(function (b) { if (a.id !== b.id) pairs.push({ depTerminalId: a.id, arrTerminalId: b.id, depPlandTime: date, numOfRows: 200, pageNo: 1 }); }); });
  var out = [];
  callMany(infoKey, pairs).forEach(function (items) {
    items.forEach(function (it) {
      out.push({ label: word + ' ' + (it.gradeNm || '') + ' (' + (it.depPlaceNm || '') + '→' + (it.arrPlaceNm || '') + ')', fare: Number(it.charge) || 0 });
    });
  });
  return out;
}

function trainOptions(from, to, date) {
  var stations = cached('stations', function () {
    var cities = callAll('city', {});
    var lists = callMany('sttn', cities.map(function (c) { return { cityCode: c.citycode, numOfRows: 500, pageNo: 1 }; }));
    var all = [];
    lists.forEach(function (l) { l.forEach(function (s) { all.push({ id: s.nodeid, nm: String(s.nodename || '') }); }); });
    return all;
  });
  var A = pick(stations, TRAIN_ALIAS[from] || [from], !!TRAIN_ALIAS[from], 3);
  var B = pick(stations, TRAIN_ALIAS[to] || [to], !!TRAIN_ALIAS[to], 3);
  var pairs = [];
  A.forEach(function (a) { B.forEach(function (b) { if (a.id !== b.id) pairs.push({ depPlaceId: a.id, arrPlaceId: b.id, depPlandTime: date, numOfRows: 200, pageNo: 1 }); }); });
  var out = [];
  callMany('train', pairs).forEach(function (items) {
    var rows = items.map(function (it) {
      var g = String(it.traingradename || '열차').replace(/\(.*?\)/g, '');   // "KTX-산천(A-type)" → "KTX-산천"
      if (g.indexOf('KTX-산천') === 0) g = 'KTX';
      return { g: g, dep: it.depplacename || '', arr: it.arrplacename || '', fare: Number(it.adultcharge) || 0 };
    });
    // 포털이 무궁화호·누리로에 ITX 와 같은 운임을 주는 경우가 있어 표시해 둡니다.
    var itx = {};
    rows.forEach(function (r) { if (r.g.indexOf('ITX') === 0) itx[r.fare] = 1; });
    rows.sort(function (a, b) { return (a.g.indexOf('KTX') === 0 ? 0 : 1) - (b.g.indexOf('KTX') === 0 ? 0 : 1) || a.fare - b.fare; });
    rows.forEach(function (r) {
      var doubt = /무궁화|누리로/.test(r.g) && itx[r.fare];
      out.push({ label: r.g + ' (' + r.dep + '→' + r.arr + ')' + (doubt ? ' ※요금 확인 필요' : ''), fare: r.fare });
    });
  });
  return out;
}

function airOptions(from, to, date) {
  var ports = cached('airports', function () {
    return callAll('arprt', {}).map(function (a) { return { id: a.airportId, nm: String(a.airportNm || '') }; });
  });
  var A = pick(ports, AIR_ALIAS[from] || [from], false, 2), B = pick(ports, AIR_ALIAS[to] || [to], false, 2);
  var pairs = [];
  A.forEach(function (a) { B.forEach(function (b) { if (a.id !== b.id) pairs.push({ depAirportId: a.id, arrAirportId: b.id, depPlandTime: date, numOfRows: 200, pageNo: 1 }); }); });
  var out = [];
  callMany('flight', pairs).forEach(function (items) {
    items.forEach(function (it) {
      out.push({ label: (it.airlineNm || '항공') + ' 일반석 (' + (it.depAirportNm || '') + '→' + (it.arrAirportNm || '') + ')', fare: Number(it.economyCharge) || 0 });
    });
  });
  return out;
}

// "여수(전남학생교육문화회관)" → "여수", "서울특별시" → "서울"
function cityName(s) {
  return String(s || '').replace(/\(.*?\)/g, '').replace(/(특별자치시|특별자치도|특별시|광역시|시|군)$/, '').replace(/\s/g, '');
}

// 이름이 맞는 터미널·역·공항 고르기. exact=true 이면 이름이 똑같은 것만.
function pick(list, names, exact, max) {
  var hit = list.filter(function (x) {
    return names.some(function (n) { return exact ? x.nm === n : x.nm.indexOf(n) >= 0; });
  });
  if (exact) { hit.sort(function (a, b) { return names.indexOf(a.nm) - names.indexOf(b.nm); }); return hit.slice(0, max); }
  hit.sort(function (a, b) {
    var sa = names.some(function (n) { return a.nm.indexOf(n) === 0; }) ? 0 : 1;
    var sb = names.some(function (n) { return b.nm.indexOf(n) === 0; }) ? 0 : 1;
    return sa - sb || a.nm.length - b.nm.length;
  });
  return hit.slice(0, max);
}

// 같은 항목은 하나로, 요금 0원은 빼고, 싼 순서로 최대 12개
function tidy(opts, keepOrder) {
  var seen = {}, out = [];
  opts.forEach(function (o) {
    if (!(o.fare > 0)) return;
    var k = o.label + '|' + o.fare;
    if (!seen[k]) { seen[k] = 1; out.push(o); }
  });
  if (!keepOrder) out.sort(function (a, b) { return a.fare - b.fare; });
  return out.slice(0, 40);
}

function cached(key, make) {
  var cache = CacheService.getScriptCache();
  try { var hit = cache.get(key); if (hit) return JSON.parse(hit); } catch (e) {}
  var v = make();
  try { if (v && v.length) cache.put(key, JSON.stringify(v), 21600); } catch (e2) {}  // 6시간 보관
  return v;
}

function buildUrl(path, params) {
  var key = SERVICE_KEY.indexOf('%') >= 0 ? SERVICE_KEY : encodeURIComponent(SERVICE_KEY);
  var q = ['serviceKey=' + key, '_type=json'];
  for (var k in params) q.push(k + '=' + encodeURIComponent(params[k]));
  return BASE + path + '?' + q.join('&');
}

// 응답 해석. { ok, items, msg } 로 돌려줍니다.
function parse(resp) {
  var code = resp.getResponseCode(), text = resp.getContentText();
  var j;
  try { j = JSON.parse(text); } catch (e) {
    var m = text.match(/<(returnAuthMsg|errMsg|resultMsg|cmmMsgHeader)>([^<]*)/);
    return { ok: false, items: [], msg: (m ? m[2] : text.slice(0, 80)) + ' (HTTP ' + code + ')' };
  }
  var r = j.response || j, h = r.header || {};
  if (code !== 200) return { ok: false, items: [], msg: (h.resultMsg || j.message || 'HTTP ' + code) };
  if (h.resultCode !== undefined && String(h.resultCode) !== '00' && String(h.resultCode) !== '0') {
    return { ok: false, items: [], msg: String(h.resultMsg || h.resultCode) };
  }
  var items = r.body && r.body.items && r.body.items.item;
  if (!items) return { ok: true, items: [] };
  return { ok: true, items: Array.isArray(items) ? items : [items] };
}

// 한 번 호출. 주소 후보를 차례로 시도합니다.
function callOne(key, params) {
  var cands = PATHS[key], start = GOOD[key] || 0, last = '';
  for (var i = start; i < cands.length; i++) {
    var res = parse(UrlFetchApp.fetch(buildUrl(cands[i], params), { muteHttpExceptions: true }));
    if (res.ok) { GOOD[key] = i; return res.items; }
    last = res.msg;
  }
  ERR.push(key + ': ' + last);
  return [];
}

// 목록 전체 받기(여러 쪽에 걸쳐 있으면 끝까지)
function callAll(key, params) {
  var all = [], size = 500;
  for (var page = 1; page <= 10; page++) {
    var p = { numOfRows: size, pageNo: page };
    for (var k in params) p[k] = params[k];
    var items = callOne(key, p);
    all = all.concat(items);
    if (items.length < size) break;
  }
  return all;
}

// 여러 건을 한꺼번에 호출
function callMany(key, paramsList) {
  if (!paramsList.length) return [];
  var first = callOne(key, paramsList[0]);
  var out = [first];
  if (paramsList.length > 1) {
    var path = PATHS[key][GOOD[key] || 0];
    var reqs = paramsList.slice(1).map(function (p) { return { url: buildUrl(path, p), muteHttpExceptions: true }; });
    UrlFetchApp.fetchAll(reqs).forEach(function (resp) { out.push(parse(resp).items); });
  }
  return out;
}
